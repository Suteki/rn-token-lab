import React from 'react';
import { Button, StyleSheet, Text, TextInput, View } from 'react-native';
import { Controller, useForm } from 'react-hook-form';
import { zodResolver } from '@hookform/resolvers/zod';
import { NativeStackScreenProps } from '@react-navigation/native-stack';
import { RootStackParamList } from '../navigation/RootNavigator';
import { TaskDraft, taskDraftSchema } from '../domain/task';
import { usePreferences } from '../stores/preferences';
type Props = NativeStackScreenProps<RootStackParamList, 'NewTask'>;
export function NewTaskScreen({ navigation }: Props) {
  const addTask = usePreferences(state => state.addTask);
  const {
    control,
    handleSubmit,
    formState: { errors },
  } = useForm<TaskDraft>({
    resolver: zodResolver(taskDraftSchema),
    defaultValues: { title: '', priority: 'normal' },
  });
  const submit = handleSubmit(draft => {
    addTask(draft);
    navigation.goBack();
  });
  return (
    <View style={styles.page}>
      <Text style={styles.label}>Title</Text>
      <Controller
        control={control}
        name="title"
        render={({ field: { value, onChange, onBlur } }) => (
          <TextInput
            value={value}
            onChangeText={onChange}
            onBlur={onBlur}
            placeholder="What needs doing?"
            placeholderTextColor="#94a3b8"
            style={styles.input}
            accessibilityLabel="Task title"
          />
        )}
      />
      {errors.title ? (
        <Text accessibilityRole="alert" style={styles.error}>
          {errors.title.message}
        </Text>
      ) : null}
      <Controller
        control={control}
        name="priority"
        render={({ field: { value, onChange } }) => (
          <Button
            title={`Priority: ${value}`}
            onPress={() => onChange(value === 'normal' ? 'high' : 'normal')}
          />
        )}
      />
      <Button title="Save on this device" onPress={() => submit()} />
      <Text style={styles.note}>
        This form creates a local draft and never posts personal data to the
        demo API.
      </Text>
    </View>
  );
}
const styles = StyleSheet.create({
  page: { flex: 1, padding: 24 },
  label: { color: '#f8fafc', fontSize: 18, marginBottom: 12 },
  input: {
    backgroundColor: '#1e293b',
    color: '#f8fafc',
    borderRadius: 12,
    padding: 16,
    marginBottom: 16,
  },
  error: { color: '#fca5a5', marginBottom: 12 },
  note: { color: '#94a3b8', marginTop: 28, lineHeight: 22 },
});

// FILE: src/screens/TaskDetailsScreen.tsx
import React from 'react';
import { Button, StyleSheet, Text, View } from 'react-native';
import { NativeStackScreenProps } from '@react-navigation/native-stack';
import { RootStackParamList } from '../navigation/RootNavigator';
import { useTasks } from '../hooks/useTasks';
import { usePreferences } from '../stores/preferences';
type Props = NativeStackScreenProps<RootStackParamList, 'TaskDetails'>;
export function TaskDetailsScreen({ route }: Props) {
  const { data = [] } = useTasks();
  const { localTasks, favoriteIds, toggleFavorite } = usePreferences();
  const task = [...localTasks, ...data].find(
    item => item.id === route.params.taskId,
  );
  if (!task) {
    return (
      <Text style={styles.missing}>Task not found in the current list</Text>
    );
  }
  return (
    <View style={styles.page}>
      <Text style={styles.label}>Task #{task.id}</Text>
      <Text style={styles.title}>{task.title}</Text>
      <Text style={styles.label}>{task.completed ? 'Complete' : 'Open'}</Text>
      <Button
        title={
          favoriteIds.includes(task.id) ? 'Remove favorite' : 'Add favorite'
        }
        onPress={() => toggleFavorite(task.id)}
      />
      <Text style={styles.note}>
        Favorites stay on this device. Public API tasks are read only.
      </Text>
    </View>
  );
}
const styles = StyleSheet.create({
  page: { flex: 1, padding: 24 },
  label: { color: '#a5b4fc', fontSize: 17, marginBottom: 20 },
  title: { color: '#f8fafc', fontSize: 28, lineHeight: 38, marginBottom: 24 },
  note: { color: '#94a3b8', marginTop: 28, lineHeight: 22 },
  missing: { color: '#f8fafc', padding: 24 },
});


// FILE: src/screens/TaskListScreen.tsx
import React, { useMemo, useState } from 'react';
import {
  ActivityIndicator,
  Button,
  FlatList,
  StyleSheet,
  Text,
  TextInput,
  View,
} from 'react-native';
import { NativeStackScreenProps } from '@react-navigation/native-stack';
import { RootStackParamList } from '../navigation/RootNavigator';
import { filterTasks, getTaskStats } from '../domain/task';
import { useTasks } from '../hooks/useTasks';
import { useConnectivity } from '../hooks/useConnectivity';
import { usePreferences } from '../stores/preferences';
import { TaskCard } from '../components/TaskCard';
import { ProgressRing } from '../components/ProgressRing';
type Props = NativeStackScreenProps<RootStackParamList, 'Tasks'>;

export function TaskListScreen({ navigation }: Props) {
  const { data = [], isPending, isRefetching, error, refetch } = useTasks();
  const isOnline = useConnectivity();
  const [query, setQuery] = useState('');
  const [favoritesOnly, setFavoritesOnly] = useState(false);
  const { localTasks, favoriteIds, showCompleted } = usePreferences();
  const allTasks = useMemo(() => [...localTasks, ...data], [localTasks, data]);
  const visible = filterTasks(allTasks, {
    query,
    showCompleted,
    favoriteIds,
    favoritesOnly,
  });
  const stats = getTaskStats(allTasks);
  return (
    <View style={styles.page}>
      <View style={styles.toolbar}>
        <View style={styles.summary}>
          <Text style={styles.title}>{stats.total} tasks</Text>
          <Text style={styles.hint}>
            {isOnline === false
              ? 'Offline, showing cached tasks'
              : 'Public demo API and local drafts'}
          </Text>
        </View>
        <ProgressRing percent={stats.completionPercent} />
      </View>
      <View style={styles.actions}>
        <Button title="Add" onPress={() => navigation.navigate('NewTask')} />
        <Button
          title="Settings"
          onPress={() => navigation.navigate('Settings')}
        />
        <Button
          title={favoritesOnly ? 'Show all' : 'Favorites'}
          onPress={() => setFavoritesOnly(!favoritesOnly)}
        />
      </View>
      <TextInput
        value={query}
        onChangeText={setQuery}
        placeholder="Search task titles"
        placeholderTextColor="#94a3b8"
        style={styles.search}
        accessibilityLabel="Search tasks"
      />
      {error ? (
        <Text accessibilityRole="alert" style={styles.error}>
          {error.message}
        </Text>
      ) : null}
      {isPending ? (
        <ActivityIndicator accessibilityLabel="Loading tasks" />
      ) : null}
      <FlatList
        data={visible}
        keyExtractor={task => String(task.id)}
        refreshing={isRefetching}
        onRefresh={() => refetch()}
        renderItem={({ item }) => (
          <TaskCard
            task={item}
            favorite={favoriteIds.includes(item.id)}
            onPress={() =>
              navigation.navigate('TaskDetails', { taskId: item.id })
            }
          />
        )}
        ListEmptyComponent={
          !isPending ? (
            <Text style={styles.hint}>No matching tasks</Text>
          ) : undefined
        }
      />
    </View>
  );
}
const styles = StyleSheet.create({
  page: { flex: 1, padding: 20 },
  toolbar: { flexDirection: 'row', alignItems: 'center', marginBottom: 18 },
  summary: { flex: 1 },
  title: { color: '#f8fafc', fontSize: 29, fontWeight: '700' },
  hint: { color: '#94a3b8', fontSize: 14, marginTop: 8 },
  actions: {
    flexDirection: 'row',
    justifyContent: 'space-between',
    marginBottom: 14,
  },
  search: {
    backgroundColor: '#1e293b',
    color: '#f8fafc',
    borderRadius: 12,
    padding: 14,
    marginBottom: 16,
  },
  error: { color: '#fca5a5', marginBottom: 12 },
});
