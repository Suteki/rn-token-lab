import React from 'react';
import Svg, { Circle, Text as SvgText } from 'react-native-svg';
export function ProgressRing({
  percent,
  size = 88,
}: {
  percent: number;
  size?: number;
}) {
  const value = Math.max(0, Math.min(100, percent));
  const radius = (size - 12) / 2;
  const perimeter = 2 * Math.PI * radius;
  return (
    <Svg width={size} height={size} accessibilityLabel={`${value}% complete`}>
      <Circle
        cx={size / 2}
        cy={size / 2}
        r={radius}
        stroke="#334155"
        strokeWidth={6}
        fill="none"
      />
      <Circle
        cx={size / 2}
        cy={size / 2}
        r={radius}
        stroke="#a5b4fc"
        strokeWidth={6}
        fill="none"
        strokeDasharray={`${perimeter} ${perimeter}`}
        strokeDashoffset={perimeter * (1 - value / 100)}
        rotation={-90}
        origin={`${size / 2}, ${size / 2}`}
      />
      <SvgText x="50%" y="55%" textAnchor="middle" fill="#f8fafc" fontSize={19}>
        {value}%
      </SvgText>
    </Svg>
  );
}

import { parseTasks, Task } from '../domain/task';
export const TASKS_URL = 'https://jsonplaceholder.typicode.com/todos?_limit=20';

export async function fetchTasks(signal?: AbortSignal): Promise<Task[]> {
  const response = await fetch(TASKS_URL, { signal });
  if (!response.ok) {
    throw new Error(`Task service returned HTTP ${response.status}`);
  }
  return parseTasks(await response.json());
}

import { z } from 'zod';

export const taskSchema = z.object({
  id: z.number().int().positive(),
  userId: z.number().int().positive(),
  title: z.string().trim().min(1).max(240),
  completed: z.boolean(),
});
export const taskDraftSchema = z.object({
  title: z.string().trim().min(3, 'Enter at least 3 characters').max(120),
  priority: z.enum(['normal', 'high']),
});
export type Task = z.infer<typeof taskSchema>;
export type TaskDraft = z.infer<typeof taskDraftSchema>;
export type LocalTask = Task & { priority: TaskDraft['priority'] };
export type TaskFilter = {
  query: string;
  showCompleted: boolean;
  favoritesOnly: boolean;
  favoriteIds: number[];
};

export function parseTasks(value: unknown): Task[] {
  return z.array(taskSchema).parse(value);
}

export function filterTasks(
  tasks: readonly Task[],
  filter: TaskFilter,
): Task[] {
  const query = filter.query.trim().toLocaleLowerCase();
  const favorites = new Set(filter.favoriteIds);
  return tasks.filter(task => {
    if (!filter.showCompleted && task.completed) {
      return false;
    }
    if (filter.favoritesOnly && !favorites.has(task.id)) {
      return false;
    }
    return task.title.toLocaleLowerCase().includes(query);
  });
}

export function getTaskStats(tasks: readonly Task[]) {
  const total = tasks.length;
  const completed = tasks.filter(task => task.completed).length;
  return {
    total,
    completed,
    pending: total - completed,
    completionPercent: total === 0 ? 0 : Math.round((completed / total) * 100),
  };
}

export function toggleFavorite(ids: readonly number[], id: number): number[] {
  return ids.includes(id) ? ids.filter(value => value !== id) : [...ids, id];
}

export function createLocalTask(draft: TaskDraft, nextId: number): LocalTask {
  if (!Number.isInteger(nextId) || nextId <= 0) {
    throw new Error('A local task needs a positive integer id');
  }
  const valid = taskDraftSchema.parse(draft);
  return { id: nextId, userId: 1, completed: false, ...valid };
}

import { useEffect, useState } from 'react';
import NetInfo from '@react-native-community/netinfo';
import { onlineManager } from '@tanstack/react-query';
export function useConnectivity() {
  const [isOnline, setOnline] = useState<boolean | null>(null);
  useEffect(
    () =>
      NetInfo.addEventListener(state => {
        const online =
          state.isConnected !== false && state.isInternetReachable !== false;
        setOnline(online);
        onlineManager.setOnline(online);
      }),
    [],
  );
  return isOnline;
}

import React, { useMemo, useState } from 'react';
import {
  ActivityIndicator,
  Button,
  FlatList,
  StyleSheet,
  Text,
  TextInput,
  View,
} from 'react-native';
import { NativeStackScreenProps } from '@react-navigation/native-stack';
import { RootStackParamList } from '../navigation/RootNavigator';
import { filterTasks, getTaskStats } from '../domain/task';
import { useTasks } from '../hooks/useTasks';
import { useConnectivity } from '../hooks/useConnectivity';
import { usePreferences } from '../stores/preferences';
import { TaskCard } from '../components/TaskCard';
import { ProgressRing } from '../components/ProgressRing';
type Props = NativeStackScreenProps<RootStackParamList, 'Tasks'>;

export function TaskListScreen({ navigation }: Props) {
  const { data = [], isPending, isRefetching, error, refetch } = useTasks();
  const isOnline = useConnectivity();
  const [query, setQuery] = useState('');
  const [favoritesOnly, setFavoritesOnly] = useState(false);
  const { localTasks, favoriteIds, showCompleted } = usePreferences();
  const allTasks = useMemo(() => [...localTasks, ...data], [localTasks, data]);
  const visible = filterTasks(allTasks, {
    query,
    showCompleted,
    favoriteIds,
    favoritesOnly,
  });
  const stats = getTaskStats(allTasks);
  return (
    <View style={styles.page}>
      <View style={styles.toolbar}>
        <View style={styles.summary}>
          <Text style={styles.title}>{stats.total} tasks</Text>
          <Text style={styles.hint}>
            {isOnline === false
              ? 'Offline, showing cached tasks'
              : 'Public demo API and local drafts'}
          </Text>
        </View>
        <ProgressRing percent={stats.completionPercent} />
      </View>
      <View style={styles.actions}>
        <Button title="Add" onPress={() => navigation.navigate('NewTask')} />
        <Button
          title="Settings"
          onPress={() => navigation.navigate('Settings')}
        />
        <Button
          title={favoritesOnly ? 'Show all' : 'Favorites'}
          onPress={() => setFavoritesOnly(!favoritesOnly)}
        />
      </View>
      <TextInput
        value={query}
        onChangeText={setQuery}
        placeholder="Search task titles"
        placeholderTextColor="#94a3b8"
        style={styles.search}
        accessibilityLabel="Search tasks"
      />
      {error ? (
        <Text accessibilityRole="alert" style={styles.error}>
          {error.message}
        </Text>
      ) : null}
      {isPending ? (
        <ActivityIndicator accessibilityLabel="Loading tasks" />
      ) : null}
      <FlatList
        data={visible}
        keyExtractor={task => String(task.id)}
        refreshing={isRefetching}
        onRefresh={() => refetch()}
        renderItem={({ item }) => (

            onPress={() =>

          />
        )}
        ListEmptyComponent={
          !isPending ? (
            <Text style={styles.hint}>No matching tasks</Text>

        }
      />
    </View>
  );

import { createAsyncStorage } from '@react-native-async-storage/async-storage';
import { create } from 'zustand';
import { createJSONStorage, persist } from 'zustand/middleware';
import {
  createLocalTask,
  LocalTask,
  TaskDraft,
  toggleFavorite,
} from '../domain/task';
export const preferencesStorage = createAsyncStorage('token-savings-rn');
type Preferences = {
  showCompleted: boolean;
  favoriteIds: number[];
  localTasks: LocalTask[];
  setShowCompleted: (value: boolean) => void;
  toggleFavorite: (id: number) => void;
  addTask: (draft: TaskDraft) => void;
};
export const usePreferences = create<Preferences>()(
  persist(
    (set, get) => ({
      showCompleted: true,
      favoriteIds: [],
      localTasks: [],
      setShowCompleted: showCompleted => set({ showCompleted }),
      toggleFavorite: id =>
        set({ favoriteIds: toggleFavorite(get().favoriteIds, id) }),
      addTask: draft => {
        const tasks = get().localTasks;
        const nextId = Math.max(1_000_000, ...tasks.map(task => task.id)) + 1;
        set({ localTasks: [...tasks, createLocalTask(draft, nextId)] });
      },
    }),
    {
      name: 'preferences',
      storage: createJSONStorage(() => preferencesStorage),
      version: 1,
      partialize: state => ({
        showCompleted: state.showCompleted,
        favoriteIds: state.favoriteIds,
        localTasks: state.localTasks,
      }),
    },
  ),
);
